import ast
import subprocess
import sys

ALLOWED_STDLIB = {
    "math", "dataclasses", "typing", "collections", "itertools",
    "functools", "statistics", "json", "re", "datetime", "enum",
    "hashlib", "sqlite3", "pathlib", "os", "sys"
}

def audit_python_ast_and_runtime(code_str: str) -> bool:
    tree = ast.parse(code_str)
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for alias in node.names:
                if alias.name.split(".")[0] not in ALLOWED_STDLIB:
                    raise ValueError(f"External dependency blocked: {alias.name}")
    proc = subprocess.run([sys.executable, "-X", "utf8", "-c", code_str], capture_output=True, timeout=10)
    return proc.returncode == 0
