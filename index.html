def audit_python_ast_and_runtime(code_str: str) -> bool:
    tree = ast.parse(code_str)
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for alias in node.names:
                if alias.name.split(".")[0] not in ALLOWED_STDLIB:
                    raise ValueError(f"External dependency blocked: {alias.name}")

    return proc.returncode == 0

def decode_hwp5_para_text_stdlib(payload: bytes) -> str:
    n_u16 = len(payload) // 2
    codes = struct.unpack_from(f"<{n_u16}H", payload, 0)
    out, idx = [], 0
    while idx < n_u16:
        ch = codes[idx]
        if ch in HWP5_8WCHAR_CONTROLS:
            if ch == 9:
                out.append("\t")
            idx += 8  # 16바이트 이진 좌표 페이로드를 통째로 건너뛰어 깨짐 문자(\ufffd) 0건 보장
            continue
        if ch in (10, 13):
            out.append("\n")
        elif ch >= 32 and not (0xD800 <= ch <= 0xDFFF):
            out.append(chr(ch))
        idx += 1
    return "".join(out).strip()
